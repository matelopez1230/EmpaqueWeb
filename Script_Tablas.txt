-- EMPAQUES
CREATE TABLE [dbo].[EmpWebEmpaques](
	[IdEmpaque] [int] IDENTITY(1,1) NOT NULL,
	[NumeroEmpaque] [varchar](20) NOT NULL,
	[Cliente] [varchar](100) NULL,
	[DireccionEnvio] [varchar](255) NULL,
	[FechaCreacion] [datetime] DEFAULT (getdate()),
	[UsuarioCreacion] [int] NULL,
	[Estado] [varchar](20) DEFAULT ('PENDIENTE'),
	[FechaCompletado] [datetime] NULL,
	[ZonaEnvio] [varchar](100) NULL,
	[FechaPalletizado] [datetime] NULL,
	[UsuarioPalletizado] [int] NULL,
	[ControlCompletado] [bit] DEFAULT ((0)),
    CONSTRAINT [PK_EmpWebEmpaques] PRIMARY KEY ([IdEmpaque]),
    CONSTRAINT [UQ_EmpWebEmpaques_Numero] UNIQUE ([NumeroEmpaque])
);

-- EMPAQUE DETALLE
CREATE TABLE [dbo].[EmpWebEmpaqueDetalle](
	[IdEmpaqueDetalle] [int] IDENTITY(1,1) NOT NULL,
	[IdEmpaque] [int] NULL,
	[NroCbt] [varchar](20) NULL,
	[CodArt] [varchar](20) NULL,
	[CantidadRequerida] [decimal](18, 4) NULL,
	[CantidadPickeada] [decimal](18, 4) DEFAULT ((0)),
	[Estado] [varchar](20) DEFAULT ('PENDIENTE'),
    CONSTRAINT [PK_EmpWebEmpaqueDetalle] PRIMARY KEY ([IdEmpaqueDetalle])
);

-- PALLET EMPAQUE
CREATE TABLE [dbo].[EmpWebPalletEmpaque](
	[IdPallet] [int] IDENTITY(1,1) NOT NULL,
	[IdEmpaque] [int] NOT NULL,
	[NumeroPallet] [int] NOT NULL,
	[UsuarioAsignado] [int] NULL,
	[FechaAsignacion] [datetime] NULL,
	[Estado] [varchar](20) DEFAULT ('PENDIENTE'),
	[ControlEstado] [varchar](20) DEFAULT ('PENDIENTE'),
    PRIMARY KEY ([IdPallet])
);

-- PALLET DETALLE
CREATE TABLE [dbo].[EmpWebPalletDetalle](
	[IdPalletDetalle] [int] IDENTITY(1,1) NOT NULL,
	[IdPallet] [int] NOT NULL,
	[IdEmpaqueDetalle] [int] NOT NULL,

	[FechaAsignacion] [datetime] NULL,

);
